cmake_minimum_required(VERSION 3.20)
project(iqdata_native C)
set(CMAKE_C_STANDARD 11)
option(ENABLE_SANITIZERS "Address and undefined behavior sanitizers" ON)
if(ENABLE_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
enable_testing()
add_executable(test_health test_health.c ../main/iq_health.c)
target_include_directories(test_health PRIVATE ../main)
add_test(NAME startup_health COMMAND test_health)
set(PORT "${CMAKE_CURRENT_LIST_DIR}/..")
add_library(iqcore STATIC ${PORT}/main/iq_points.c ${PORT}/main/iq_decode.c
    ${PORT}/main/iq_diagnostics.c ${PORT}/main/iq_meter_trace.c ${PORT}/main/iq_observe.c ${PORT}/main/iq_model.c ${PORT}/main/iq_transport.c ${PORT}/main/iq_config.c ${PORT}/main/iq_uf2.c
    ${PORT}/third_party/cJSON/cJSON.c)
target_include_directories(iqcore PUBLIC ${PORT}/main ${PORT}/third_party/cJSON)
target_link_libraries(iqcore PUBLIC m)
target_compile_options(iqcore PRIVATE -Wall -Wextra -Werror)
set_source_files_properties(${PORT}/third_party/cJSON/cJSON.c
    PROPERTIES COMPILE_OPTIONS "-Wno-deprecated-declarations")
include(${PORT}/components/bacnet_stack/bacnet_sources.cmake)
add_library(iqbacnet STATIC ${BACNET_SOURCES} ${PORT}/main/gateway_bacnet.c ${PORT}/main/bacnet_restart.c ${PORT}/main/bip_port.c)
target_include_directories(iqbacnet PUBLIC ${PORT}/main ${PORT}/components/bacnet_stack/include ${BACNET_SRC})
target_compile_definitions(iqbacnet PUBLIC ${BACNET_DEFINITIONS})
target_link_libraries(iqbacnet PUBLIC iqcore)
add_executable(iq_oracle oracle.c)
target_link_libraries(iq_oracle PRIVATE iqcore)
add_executable(iq_bacnet_server bacnet_server.c)
target_link_libraries(iq_bacnet_server PRIVATE iqbacnet)
add_executable(pico_usb ../../pico-live/usb_tx.c ../../pico-live/test_usb_tx.c)
add_executable(pico_writer_arm ../../pico-live/test_writer_arm.c)
add_test(NAME pico_usb COMMAND pico_usb)
add_test(NAME pico_writer_arm COMMAND pico_writer_arm)
add_executable(uf2_guard uf2_guard.c)
target_link_libraries(uf2_guard PRIVATE iqcore)
add_test(NAME uf2_guard COMMAND uf2_guard)
add_executable(rolling_windows rolling_windows.c)
target_link_libraries(rolling_windows PRIVATE iqcore)
add_test(NAME rolling_windows COMMAND rolling_windows)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME port_parity COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_LIST_DIR}/test_port.py $<TARGET_FILE:iq_oracle> -v)

add_test(NAME secure_client COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/test_secure_client.py -v)

add_executable(test_diagnostics test_diagnostics.c)
target_link_libraries(test_diagnostics PRIVATE iqcore)
add_test(NAME diagnostic_ring COMMAND test_diagnostics)
add_executable(test_observe test_observe.c)
target_link_libraries(test_observe PRIVATE iqcore)
add_test(NAME passive_observation COMMAND test_observe)

add_executable(test_identity test_identity.c)
