cmake_minimum_required(VERSION 3.13)
# Connected Pico 2 W independently verified by silicon/flash identification
# and a successful CYW43 identification probe. Fallback headers remain portable.
set(PICO_BOARD_HEADER_DIRS ${CMAKE_CURRENT_LIST_DIR}/boards)
set(PICO_BOARD iqdata_rp2350_common CACHE STRING "Verified RP2350-A, 4MiB flash")
if(NOT PICO_BOARD MATCHES "^(iqdata_rp2350_common|pico2|pico2_w)$")
  message(FATAL_ERROR "This firmware requires the verified RP2350-A target")
endif()
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(iqdata_pico_live C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(iqdata_pico_live main.c engine.c live_pio.c usb_tx.c)
pico_generate_pio_header(iqdata_pico_live ${CMAKE_CURRENT_LIST_DIR}/live_transport.pio)
target_link_libraries(iqdata_pico_live pico_stdlib pico_unique_id pico_multicore hardware_watchdog hardware_pio)
target_compile_options(iqdata_pico_live PRIVATE -O2 -Wall -Wextra -Werror)
target_compile_definitions(iqdata_pico_live PRIVATE LIVE_BOARD="${PICO_BOARD}" PICO_STDIO_USB_ENABLE_RESET_VIA_VENDOR_INTERFACE=1 PICO_STDIO_USB_ENABLE_RESET_VIA_BAUD_RATE=0 PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0 PICO_STDIO_USB_STDOUT_TIMEOUT_US=100000)
pico_enable_stdio_usb(iqdata_pico_live 1)
pico_enable_stdio_uart(iqdata_pico_live 0)
pico_set_program_name(iqdata_pico_live "iqdata-pico-live")
pico_set_program_version(iqdata_pico_live "0.4.8")
pico_add_extra_outputs(iqdata_pico_live)
